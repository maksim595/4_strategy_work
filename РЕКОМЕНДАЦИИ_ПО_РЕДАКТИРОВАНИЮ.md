## 📁 1. Работа с файлами

### ❌ НЕЛЬЗЯ

```powershell
# PowerShell без указания кодировки
Set-Content file.js -Value $content

# PowerShell с экранированием
$c = $c -replace '${variable}', 'new'

# Частое копирование с заменой
copy file.js file.js /Y  # Multiple times
```

### ✅ НУЖНО

```powershell
# PowerShell с UTF-8
Set-Content file.js -Value $content -Encoding UTF8
```

```javascript
// Node.js для сложных замен
const fs = require('fs');
let content = fs.readFileSync('file.js', 'utf8');
content = content.replace('old', 'new');
fs.writeFileSync('file.js', content, 'utf8');

// Резервная копия ПЕРЕД изменениями
copy file.js file.js.bak
```

## 🔍 2. Поиск и замена

### ❌ НЕЛЬЗЯ

```javascript
// Замена по полному совпадению (может не найтись из-за пробелов)
content.replace(`function foo() {
    const x = 1;
}`, newFunction);

// PowerShell regex с ${}
$c = $c -replace 'quantity=${params}', 'qty'
```

### ✅ НУЖНО

```javascript
// Поиск по уникальному идентификатору + замена по строкам
const lines = content.split('\n');
for (let i = 0; i < lines.length; i++) {
    if (lines[i].includes('function foo()')) {
        lines[i+1] = '    var x = 1;';  // Конкретная строка
        lines[i+2] = '    return x;';
        break;
    }
}
content = lines.join('\n');

// Node.js строки без интерполяции
content.replace('quantity=${params}', 'qty');
```

## 📊 3. Работа со счётчиками { }

### ❌ НЕЛЬЗЯ

```javascript
// Определение конца функции по первой }
for (let i = start; i < lines.length; i++) {
    if (lines[i].includes('}')) {
        end = i;  // ← Это может быть } от if/for
        break;
    }
}
```

### ✅ НУЖНО

```javascript
// Подсчёт всех { и }
let braceCount = 0;
for (let i = start; i < lines.length; i++) {
    for (const char of lines[i]) {
        if (char === '{') braceCount++;
        if (char === '}') braceCount--;
    }
    if (braceCount === 0 && lines[i].includes('}')) {
        end = i;  // ← Теперь это точно конец функции
        break;
    }
}
```

## 🔤 4. Кодировка и невидимые символы

### ❌ НЕЛЬЗЯ

```javascript
// Сравнение строк с возможными невидимыми символами
if (line === 'const decimals = step.toString().includes') {
    // Может не сработать из-за NBSP/Zero Width Space
}
```

### ✅ НУЖНО

```javascript
// Поиск по части строки
if (line.includes('const decimals') && line.includes('includes')) {
    // Надёжнее
}

// Или нормализация перед сравнением
const normalized = line.replace(/\u00A0/g, ' ');  // NBSP → пробел
```

## 🛠 5. const vs var в критическом коде

### ❌ НЕЛЬЗЯ

```javascript
// const в функциях, которые обрабатывают пользовательские данные
function normalizeQuantity(qty, stepSize) {
    const step = parseFloat(stepSize);  // Риск при Unicode-проблемах
    const decimals = step.toString().includes('.') ? ...
}
```

### ✅ НУЖНО

```javascript
// var в функциях округления/нормализации
function normalizeQuantity(qty, stepSize) {
    var step = parseFloat(stepSize);
    var stepStr = stepSize.toString();
    var decimals = 0;
    if (stepStr.indexOf('.') !== -1) {  // indexOf надёжнее includes
        decimals = stepStr.split('.')[1].length;
    }
    var fixed = (Math.floor(qty / step) * step).toFixed(decimals);
    return fixed;
}
```

## 📝 6. Шаблонные строки

### ❌ НЕЛЬЗЯ

```javascript
// Сложная интерполяция в replace()
content.replace(`const qs = \`timestamp=\${ts}\``, newCode);
// ← Вложенные ` могут сломать парсинг
```

### ✅ НУЖНО

```javascript
// Обычные строки для replace()
content.replace('const qs = "timestamp=" + ts', newCode);

// Или String.raw для точного совпадения
content.replace(String.raw`const qs = \`timestamp=\${ts}\``, newCode);
```

## 📋 8. Структура скриптов для редактирования

### Шаблон безопасного скрипта

```javascript
// fix_something.js
const fs = require('fs');

console.log('=== Описание правки ===\n');

// 1. Чтение
let content = fs.readFileSync('file.js', 'utf8');
console.log('✓ Файл прочитан');

// 2. Проверка перед изменением
if (!content.includes('expected_marker')) {
    console.log('✗ Ожидаемый маркер не найден!');
    process.exit(1);
}

// 3. Изменение (построчно или replace)
content = content.replace('old', 'new');

// 4. Проверка после изменения
if (!content.includes('new')) {
    console.log('✗ Изменение не применено!');
    process.exit(1);
}

// 5. Запись с UTF-8
fs.writeFileSync('file.js', content, 'utf8');
console.log('✓ Файл сохранён в UTF-8');

// 6. Для JS — проверка синтаксиса
const { execSync } = require('child_process');
try {
    execSync('node -c file.js', { stdio: 'inherit' });
    console.log('✓ Синтаксис OK');
} catch (e) {
    console.log('✗ Синтаксическая ошибка!');
    process.exit(1);
}

console.log('\n=== ГОТОВО ===');
```

## 🚨 10. Красные флаги (немедленно остановить!)

| Ситуация | Действие |
|----------|----------|
| Файл стал 1 строкой | ❌ Восстановить из `.bak` |
| `node -c` выдаёт ошибку | ❌ Восстановить из `.bak` |
| `{` и `}` не совпадают | ❌ Проверить счётчиком |
| Кракозябры в консоли | ❌ Пересохранить в UTF-8 |
| `replace()` не находит строку | ❌ Проверить невидимые символы |
| PowerShell меняет `${}` | ❌ Перейти на Node.js |

## 🔗 13. Полезные команды

### Windows PowerShell

```powershell
# Проверка кодировки файла
Get-Content file.js -Encoding UTF8 | Select-Object -First 5

# Поиск строки в файле
Select-String -Path "file.js" -Pattern "function normalizeQuantity"

# Резервная копия
Copy-Item file.js file.js.bak -Force
```

### Node.js

```bash
# Проверка синтаксиса
node -c server.js

# Запуск скрипта
node fix_something.js

# Установка зависимостей
npm install
```

### Git (если используется)

```bash
# Резервная копия через git
git add -A
```
